% قاب دور شکل‌ها
\newcommand{\figframe}[1]{%
	\begin{tcolorbox}[colback=white,colframe=blue!70!black,boxrule=0.9pt,arc=2pt,
		left=10pt,right=10pt,top=8pt,bottom=8pt,boxsep=2pt,
		halign=center,width=\linewidth]
		#1
	\end{tcolorbox}%
}

\newcommand{\imgbox}[2][0.92\textwidth]{%
	\begin{tcolorbox}[colback=white,colframe=black!75,boxrule=0.8pt,arc=1pt,
		left=2pt,right=2pt,top=2pt,bottom=2pt,boxsep=1pt,halign=center]
		\includegraphics[width=#1]{#2}%
	\end{tcolorbox}%
}
